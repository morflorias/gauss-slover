# Пример системы линейных уравнений 4x4
# Система уравнений:
#   1*x1 + 2*x2 + 3*x3 + 4*x4 = 30
#   2*x1 + 1*x2 + 1*x3 + 1*x4 = 10
#   3*x1 + 1*x2 + 2*x3 + 1*x4 = 15
#   4*x1 + 1*x2 + 1*x3 + 2*x4 = 20
#
# Ожидаемое решение: x1 = 1, x2 = 2, x3 = 3, x4 = 4

1 2 3 4 30
2 1 1 1 10
3 1 2 1 15
4 1 1 2 20
